package com.momos.millenairejeimm.jei.type.layout;

import net.minecraft.client.gui.Font;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.network.chat.Component;

// region 建筑分类链接框布局
public class BuildingLinkLayout {
    // [声明缺少的参数对象] 适用村庄、居住村民、变体与消耗占位
    public final JeiLinkBox<Object> villageTypeLinks = new JeiLinkBox<>("适用村庄类型", 1);
    public final JeiLinkBox<Object> villagerLinks = new JeiLinkBox<>("居住/工作村民", 1);

    // 页签 1：变体，页签 2：升级消耗/总消耗
    public final JeiLinkBox<Object> tabBox = new JeiLinkBox<>("[页签] 变体 / 升级总消耗", 2);

    private int activeTab = 0; // 0: 变体, 1: 升级消耗

    public void render(GuiGraphics guiGraphics, Font font, int startY) {
        int currentY = startY;

        // (1) 顺位 1：关联所属村庄
        villageTypeLinks.render(guiGraphics, font, 2, currentY, 172);
        currentY += villageTypeLinks.getBoxHeight() + 2;

        // (2) 顺位 2：关联村民
        villagerLinks.render(guiGraphics, font, 2, currentY, 172);
        currentY += villagerLinks.getBoxHeight() + 2;

        // (3) 顺位 3：两级可翻页页签 (变体 / 消耗)
        tabBox.render(guiGraphics, font, 2, currentY, 172);

        // 渲染简易页签切换标签按钮
        guiGraphics.drawString(font, Component.literal("§b[变体] §7[升级]"), 125, currentY + 3, 0xFFFFFF, false);
    }
}
// endregion 建筑分类链接框布局
